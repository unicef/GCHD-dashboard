# Dependencies for the El Nino precipitation scripts.
#
# These live in their OWN virtualenv (scripts/el_nino/.venv), separate from the
# dashboard's root .venv: the GRIB/raster stack (eccodes, cfgrib, rasterio) is
# large, pulls its own GDAL and PROJ binaries, and nothing the app serves needs
# it. Keeping them apart means a GDAL/numpy upgrade here cannot break the app.
#
# Setup:
#     py -3.11 -m venv scripts/el_nino/.venv
#     scripts/el_nino/.venv/Scripts/python -m pip install -r scripts/el_nino/requirements.txt
#
# Direct dependencies only, pinned to what was verified working on
# Windows / Python 3.11. Run `pip freeze` if you need the full transitive set.

# CDS download
cdsapi==0.7.7

# GRIB reading — eccodes ships its own binary, so no system ecCodes install
cfgrib==0.9.15.1
eccodes==2.48.0

# arrays / analysis
xarray==2026.7.0
numpy==2.4.6
pandas==3.0.6

# GeoTIFF writing — rasterio bundles GDAL 3.10 and PROJ 9
rioxarray==0.19.0
rasterio==1.4.4

# Earth Engine: geeup does the upload (no GCS bucket in this project),
# earthengine-api is used to pre-create the image collections
earthengine-api==1.7.45
geeup==2.0.0
